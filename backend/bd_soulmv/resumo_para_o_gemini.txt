SELECT 
    fat.dt_competencia AS mes_competencia,
    ate.cd_atendimento AS atendimento,
    reg.nr_guia AS numero_aih,
    pac.nm_paciente AS paciente,
    TO_CHAR(ate.dt_atendimento, 'DD/MM/YYYY') AS data_internacao,
    TO_CHAR(ate.dt_alta, 'DD/MM/YYYY') AS data_alta,
    
    -- Dados do Procedimento / Exame / OPME
    it.cd_procedimento AS codigo_sus,
    ps.ds_procedimento AS descricao_procedimento,
    it.qt_lancamento AS quantidade,
    
    -- Quem executou (Pode ser o hospital, um médico ou clínica terceirizada)
    it.cd_prestador AS cod_prestador_executante,
    
    -- Divisão Financeira Oficial do SUS (SH = Hospital | SP = Profissional)
    NVL(it.vl_sh, 0) AS valor_hospitalar,
    NVL(it.vl_sp, 0) AS valor_profissional,
    (NVL(it.vl_sh, 0) + NVL(it.vl_sp, 0)) AS valor_total_item

FROM DBAMV.REG_FAT reg
INNER JOIN DBAMV.ATENDIME ate ON ate.cd_atendimento = reg.cd_atendimento
INNER JOIN DBAMV.PACIENTE pac ON pac.cd_paciente = ate.cd_paciente
LEFT JOIN DBAMV.REMESSA_FATURA rem ON rem.cd_remessa = reg.cd_remessa
LEFT JOIN DBAMV.FATURA fat ON fat.cd_fatura = rem.cd_fatura

-- Tabela Filha com os Detalhes Financeiros e Clínicos
INNER JOIN DBAMV.ITREG_FAT it ON it.cd_reg_fat = reg.cd_reg_fat
LEFT JOIN DBAMV.PROCEDIMENTO_SUS ps ON ps.cd_procedimento = it.cd_procedimento

WHERE 
    -- 1. Destrava a segurança informando a empresa do Hospital
    ate.cd_multi_empresa = 1
    
    -- 2. Filtra a competência que você quer extrair (Ex: Janeiro a Março)
    AND fat.dt_competencia >= TO_DATE('01/01/2026', 'DD/MM/YYYY')
    AND fat.dt_competencia <= TO_DATE('31/03/2026 23:59:59', 'DD/MM/YYYY HH24:MI:SS')
    
    -- Opcional: Se quiser filtrar apenas contas já FECHADAS e prontas para envio
    -- AND reg.sn_fechada = 'S'
    
ORDER BY 
    fat.dt_competencia DESC, 
    pac.nm_paciente ASC;

SELECT 
    P.CD_PACIENTE AS codigo_mv,
    P.NM_PACIENTE AS nome_paciente,
    P.NM_SOCIAL_PACIENTE AS nome_social,
    P.NR_CPF AS cpf,
    P.NR_CNS AS cns,
    P.DT_NASCIMENTO AS data_nascimento,
    P.TP_SEXO AS sexo,
    P.NM_MAE AS nome_mae,
    -- TO_CHAR transforma o número em texto para não dar erro na junção
    NVL(TO_CHAR(P.NR_DDD_CELULAR), '') || NVL(TO_CHAR(P.NR_CELULAR), '') AS celular,
    P.EMAIL AS email,
    P.DS_ENDERECO || ', ' || NVL(TO_CHAR(P.NR_ENDERECO), 'S/N') AS endereco_completo,
    P.NM_BAIRRO AS bairro,
    P.NR_CEP AS cep,
    P.CD_CIDADE AS codigo_ibge_cidade
FROM DBAMV.PACIENTE P
WHERE P.DT_CADASTRO >= TO_DATE('01/01/2023', 'DD/MM/YYYY') 
   OR P.DT_ULTIMA_ATUALIZACAO >= TO_DATE('01/01/2023', 'DD/MM/YYYY');



SELECT 
    A.CD_ATENDIMENTO AS atendimento_mv,
    A.CD_PACIENTE AS codigo_mv_paciente,
    P.NM_PACIENTE AS nome_paciente,
    A.DT_ATENDIMENTO AS data_internacao,
    A.HR_ATENDIMENTO AS hora_internacao,
    A.DT_ALTA AS data_alta,
    A.CD_LEITO AS leito_atual,
    A.NR_AUTORIZACAO_GESTOR AS numero_aih, -- A Chave de Ouro para cruzar com SISREG!
    A.CD_CID AS cid_principal,
    A.CD_PRESTADOR AS medico_responsavel,
    A.SN_INTERNADO AS status_internado, -- 'S' ou 'N'
    A.SN_OBITO AS indicativo_obito
FROM DBAMV.ATENDIME A
INNER JOIN DBAMV.PACIENTE P ON P.CD_PACIENTE = A.CD_PACIENTE
-- Filtro: 1 é geralmente o código do convênio SUS no MV (verifique o código exato no seu hospital)
WHERE A.CD_CONVENIO = 1 
  AND A.TP_ATENDIMENTO = 'I' -- 'I' para Internação
  AND A.DT_ATENDIMENTO >= TO_DATE('01/01/2024', 'DD/MM/YYYY');



SELECT

    V.DS_CON_PLA AS plano_sus,

FROM DBAMV.V_R_ATENDIMENTOS_CONVENIO V

  AND V.DT_ATENDIMENTO BETWEEN TO_DATE('01/01/2024', 'DD/MM/YYYY') AND TO_DATE('31/01/2024', 'DD/MM/YYYY');

SELECT 
    TO_CHAR(fat.dt_competencia, 'MM/YYYY') AS mes_competencia,
    it.cd_prestador AS cod_prestador,
    it.cd_procedimento AS codigo_sus,
    ps.ds_procedimento AS descricao_procedimento,
    
    -- Somatórios (Agrupamento)
    SUM(it.qt_lancamento) AS quantidade_total,
    SUM(NVL(it.vl_sh, 0)) AS total_receita_hospitalar,
    SUM(NVL(it.vl_sp, 0)) AS total_receita_profissional,
    SUM(NVL(it.vl_sh, 0) + NVL(it.vl_sp, 0)) AS valor_total_faturado

FROM DBAMV.REG_FAT reg
INNER JOIN DBAMV.ATENDIME ate ON ate.cd_atendimento = reg.cd_atendimento
LEFT JOIN DBAMV.REMESSA_FATURA rem ON rem.cd_remessa = reg.cd_remessa
LEFT JOIN DBAMV.FATURA fat ON fat.cd_fatura = rem.cd_fatura
INNER JOIN DBAMV.ITREG_FAT it ON it.cd_reg_fat = reg.cd_reg_fat
LEFT JOIN DBAMV.PROCEDIMENTO_SUS ps ON ps.cd_procedimento = it.cd_procedimento

WHERE 
    -- 1. Destrava a segurança da Empresa
    ate.cd_multi_empresa = 1
    
    -- 2. Filtro de Período (Ex: Primeiro Trimestre)
    AND fat.dt_competencia >= TO_DATE('01/01/2026', 'DD/MM/YYYY')
    AND fat.dt_competencia <= TO_DATE('31/03/2026 23:59:59', 'DD/MM/YYYY HH24:MI:SS')

GROUP BY 
    TO_CHAR(fat.dt_competencia, 'MM/YYYY'),
    it.cd_prestador,
    it.cd_procedimento,
    ps.ds_procedimento

-- Ordena do procedimento que deu mais dinheiro para o que deu menos
ORDER BY 
    mes_competencia DESC, 
    valor_total_faturado DESC;
