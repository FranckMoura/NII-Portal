function exportar(){
	var form = document.formulario;
	form.desc_solic.value = form.desc_solic.value.toUpperCase();
	// Validar data de solicitacao, campo de preenchimento obrigatorio
	if(form.dtaIniSolic.value.length == 0){
		alert('Informe a data inicial da solicitacao!');
		form.dtaIniSolic.focus();
		return false;
	}
	else {
		if(form.dtaFimSolic.value.length == 0){
			alert('Informe a data final da solicitacao!');
			form.dtaFimSolic.focus();
			return false;
		}
		if(!ValidateDateRange(form.dtaIniSolic.value,form.dtaFimSolic.value,31)){
			alert('Periodo invalido!');
			return false;
		}
	}
	// Validar data de reserva
	if(form.dtaIniReserva.value.length != 0){
		if(form.dtaFimReserva.value.length == 0){
			alert('Informe a data final da reserva!');
			form.dtaFimReserva.focus();
			return false;
		}
		if(!ValidateDateRange(form.dtaIniReserva.value,form.dtaFimReserva.value,31)){
			alert('Periodo invalido!');
			return false;
		}
	}
	// Validar data de internacao
	if(form.dtaIniInternacao.value.length != 0){
		if(form.dtaFimInternacao.value.length == 0){
			alert('Informe a data final da internacao!');
			form.dtaFimInternacao.focus();
			return false;
		}
		if(!ValidateDateRange(form.dtaIniInternacao.value,form.dtaFimInternacao.value,31)){
			alert('Periodo de internacao invalido!');
			return false;
		}
	}
	// Tratamento p/ qdo nao houver nada marcado para exportacao
	form.contador.value = getQtdItensSelecionados();
	if(getQtdItensSelecionados() == 0) {
		alert('Informe pelo menos um item para exportacao!');
		return false;
	}
	form.submit();
}

function getQtdItensSelecionados()
{
    var frm = document.formulario;
    var len = frm.elements.length;
    var i = 0;
    var count = 0;

    for( i = 0 ; i < len ; i++ )
    {
        if( frm.elements[i].type == 'checkbox' )
        {
            if( frm.elements[i].checked )
            {
                count++;
            }
        }
    }

    return count;
}
